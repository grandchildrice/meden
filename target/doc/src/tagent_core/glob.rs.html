1//! Scope matching.
2//!
3//! A grant's scope is a pattern over Unicode scalar values where `*` matches
4//! any sequence (including the empty one and `/`), `?` matches exactly one
5//! character and every other character matches itself. There are no
6//! character classes and no escapes; the vocabulary is small on purpose so
7//! that the Lean model (`TAgentSpec.Glob`) can define the same relation by
8//! structural recursion and the two can be compared exhaustively on small
9//! alphabets.
10//!
11//! The implementation is the classic iterative two-pointer algorithm with a
12//! single backtrack point, so it uses constant extra space and cannot
13//! overflow the stack on adversarial patterns.
14
15/// Does `pattern` match `text`?
16pub fn glob_match(pattern: &str, text: &str) -> bool {
17    let p: Vec<char> = pattern.chars().collect();
18    let t: Vec<char> = text.chars().collect();
19    glob_match_chars(&p, &t)
20}
21
22/// [`glob_match`] on already decoded characters.
23pub fn glob_match_chars(p: &[char], t: &[char]) -> bool {
24    let mut pi = 0usize; // index into pattern
25    let mut ti = 0usize; // index into text
26    let mut star: Option<(usize, usize)> = None; // (pattern idx after '*', text idx)
27
28    while ti < t.len() {
29        match (p.get(pi), t.get(ti)) {
30            (Some('*'), _) => {
31                // Record the backtrack point and try to match the rest of
32                // the pattern against the rest of the text.
33                pi = pi.saturating_add(1);
34                star = Some((pi, ti));
35            }
36            (Some(pc), Some(tc)) if *pc == '?' || pc == tc => {
37                pi = pi.saturating_add(1);
38                ti = ti.saturating_add(1);
39            }
40            _ => match star {
41                // Let the last '*' absorb one more character.
42                Some((spi, sti)) => {
43                    let next = sti.saturating_add(1);
44                    pi = spi;
45                    ti = next;
46                    star = Some((spi, next));
47                }
48                None => return false,
49            },
50        }
51    }
52    // Only trailing stars may remain.
53    while p.get(pi) == Some(&'*') {
54        pi = pi.saturating_add(1);
55    }
56    pi == p.len()
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62    use proptest::prelude::*;

64    /// The reference definition by structural recursion, mirroring the Lean
65    /// model. Used only in tests.
66    fn reference(p: &[char], t: &[char]) -> bool {
67        match (p.split_first(), t.split_first()) {
68            (None, None) => true,
69            (Some(('*', pr)), _) => {
70                reference(pr, t) || t.split_first().is_some_and(|(_, tr)| reference(p, tr))
71            }
72            (None, Some(_)) | (Some(_), None) => false,
73            (Some(('?', pr)), Some((_, tr))) => reference(pr, tr),
74            (Some((pc, pr)), Some((tc, tr))) => pc == tc && reference(pr, tr),
75        }
76    }
77
78    #[test]
79    fn basics() {
80        assert!(glob_match("", ""));
81        assert!(!glob_match("", "a"));
82        assert!(glob_match("*", ""));
83        assert!(glob_match("*", "anything/at/all"));
84        assert!(glob_match("/workspace/*", "/workspace/a/b.txt"));
85        assert!(!glob_match("/workspace/*", "/home/x"));
86        assert!(glob_match("*.example.com", "mail.example.com"));
87        assert!(!glob_match("*.example.com", "example.com"));
88        assert!(glob_match("a?c", "abc"));
89        assert!(!glob_match("a?c", "ac"));
90        assert!(glob_match("a*b*c", "aXXbYYc"));
91        assert!(!glob_match("a*b*c", "aXXbYY"));
92        assert!(glob_match("**", "x"));
93        assert!(glob_match("日本*", "日本語"));
94    }
95
96    proptest! {
97        #[test]
98        fn agrees_with_reference(p in "[ab*?]{0,6}", t in "[ab]{0,6}") {
99            let pc: Vec<char> = p.chars().collect();
100            let tc: Vec<char> = t.chars().collect();
101            prop_assert_eq!(glob_match_chars(&pc, &tc), reference(&pc, &tc));
102        }
103
104        #[test]
105        fn literal_pattern_matches_only_itself(p in "[a-z/.]{0,8}", t in "[a-z/.]{0,8}") {
106            prop_assert_eq!(glob_match(&p, &t), p == t);
107        }
