1//! Policies and decisions.
2//!
3//! A policy grants capabilities (optionally scoped by a glob) and may deny
4//! capabilities outright. [`check`] compares a requirement set against a
5//! policy and returns a decision with one reason per violation. Deny always
6//! wins over grant.
7//!
8//! Specification (`TAgentSpec.Policy`):
9//!
10//! * `covers g r  :=  g.cap = r.cap ∧ (g.scope = none ∨ ∃ s, r.scope = some s ∧ glob g.scope s)`
11//! * `violates p r :=  r.cap ∈ p.denies ∨ ¬ ∃ g ∈ p.grants, covers g r`
12//! * `allowed p rs :=  ∀ r ∈ rs, ¬ violates p r`
13

19/// A granted capability, optionally restricted to scopes matching a glob.
20#[derive(Clone, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
21#[serde(from = "GrantRepr", into = "GrantRepr")]
22pub struct Grant {
23    /// The capability granted.
24    pub capability: Capability,
25    /// Glob the requirement's scope must match; `None` grants it unscoped.
26    pub scope: Option<String>,
27}
28
29impl Grant {
30    /// A grant without scope.
31    pub fn new(capability: Capability, scope: Option<&str>) -> Self {
32        Self {
33            capability,
34            scope: scope.map(str::to_owned),
35        }
36    }
37
38    /// Parse `cap` or `cap[glob]`.
39    pub fn parse(label: &str) -> Result<Self, UnknownCapability> {
40        let (capability, scope) = parse_label(label)?;
41        Ok(Self { capability, scope })
42    }
43
44    /// Does this grant cover the requirement?
45    ///
46    /// An unscoped grant covers every scope. A scoped grant covers only
47    /// scoped requirements whose scope matches the glob: an unscoped
48    /// requirement asks for the whole capability and needs an unscoped grant.
49    pub fn covers(&self, req: &Requirement) -> bool {
50        if self.capability != req.capability {
51            return false;
52        }
53        match (&self.scope, &req.scope) {
54            (None, _) => true,
55            (Some(_), None) => false,
56            (Some(pat), Some(s)) => glob_match(pat, s),
57        }
58    }
59
60    /// `cap` or `cap[glob]`.
61    pub fn label(&self) -> String {
62        match &self.scope {
63            Some(s) => format!("{}[{}]", self.capability, s),
64            None => self.capability.to_string(),
65        }
66    }
67}
68
69/// Serialized form: either the label string or an object.
70#[derive(Serialize, Deserialize)]
71#[serde(untagged)]
72enum GrantRepr {
73    Label(String),
74    Object {
75        capability: Capability,
76        #[serde(default)]
77        scope: Option<String>,
78    },
79}
80
81impl From<GrantRepr> for Grant {
82    fn from(r: GrantRepr) -> Self {
83        match r {
84            GrantRepr::Object { capability, scope } => Grant { capability, scope },
85            GrantRepr::Label(s) => match Grant::parse(&s) {
86                Ok(g) => g,
87                // An unknown label cannot grant anything: map it to a grant
88                // that covers nothing rather than failing deserialization,
89                // so that a policy file with a typo fails closed.
90                Err(_) => Grant {
91                    capability: Capability::UiControl,
92                    scope: Some(String::from("\u{0}never\u{0}")),
93                },
94            },
95        }
96    }
97}
98
99impl From<Grant> for GrantRepr {
100    fn from(g: Grant) -> Self {
101        GrantRepr::Label(g.label())
102    }
103}

105/// A named set of grants and denies.
106#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
107pub struct Policy {
108    /// Human-readable name, echoed in reasons.
109    pub name: String,
110    /// Granted capabilities.
111    #[serde(default)]
112    pub grants: Vec<Grant>,
113    /// Denied capabilities. A denied capability is never allowed, whatever
114    /// the grants say.
115    #[serde(default)]
116    pub denies: Vec<Capability>,
117}
118
119impl Policy {
120    /// Build a policy from grant labels and denied capability names.
121    pub fn from_labels(
122        name: &str,
123        grants: &[&str],
124        denies: &[&str],
125    ) -> Result<Self, UnknownCapability> {
126        Ok(Self {
127            name: name.to_owned(),
128            grants: grants
129                .iter()
130                .map(|g| Grant::parse(g))
131                .collect::<Result<_, _>>()?,
132            denies: denies.iter().map(|d| d.parse()).collect::<Result<_, _>>()?,
133        })
134    }
135
136    /// Is the capability denied?
137    pub fn denies(&self, cap: Capability) -> bool {
138        self.denies.contains(&cap)
139    }
140
141    /// Does some grant cover the requirement?
142    pub fn grants(&self, req: &Requirement) -> bool {
143        self.grants.iter().any(|g| g.covers(req))
144    }
145}

147/// The outcome of checking requirements against a policy.
148#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
149pub struct Decision {
150    /// `true` when every requirement is granted and none is denied.
151    pub allowed: bool,
152    /// One line per violated requirement, in label order. Empty when allowed.
153    #[serde(default)]
154    pub reasons: Vec<String>,
155}
156
157impl Decision {
158    /// The allowing decision.
159    pub const ALLOW: Decision = Decision {
160        allowed: true,
161        reasons: Vec::new(),
162    };
163
164    /// `!allowed`.
165    pub fn denied(&self) -> bool {
166        !self.allowed
167    }
168}
169
170/// Check a set of requirements against a policy.
171///
172/// Requirements are considered in the order of their label (then evidence),
173/// so the reasons are deterministic whatever order the caller used.
174pub fn check(requirements: &[Requirement], policy: &Policy) -> Decision {
175    let mut reqs: Vec<&Requirement> = requirements.iter().collect();
176    reqs.sort_by(|a, b| {
177        a.label()
178            .cmp(&b.label())
179            .then_with(|| a.evidence.cmp(&b.evidence))
180    });
181    reqs.dedup();
182    let mut reasons = Vec::new();
183    for req in reqs {
184        if policy.denies(req.capability) {
185            reasons.push(format!(
186                "{} is denied by policy {:?} ({})",
187                req, policy.name, req.evidence
188            ));
189        } else if !policy.grants(req) {
190            reasons.push(format!(
191                "{} is not granted by policy {:?} ({})",
192                req, policy.name, req.evidence
193            ));
194        }
195    }
196    Decision {
197        allowed: reasons.is_empty(),
198        reasons,
199    }
200}

207    fn policy() -> Policy {
208        Policy::from_labels(
209            "ws",
210            &[
211                "fs.read[/workspace/*]",
212                "net.egress[*.example.com]",
213                "env.read",
214            ],
215            &["shell.exec"],
216        )
217        .unwrap_or_default()
218    }

220    #[test]
221    fn scoped_grant_covers_matching_scope_only() {
222        let p = policy();
223        assert!(
224            check(
225                &[Requirement::new(
226                    Capability::FsRead,
227                    Some("/workspace/a.txt")
228                )],
229                &p
230            )
231            .allowed
232        );
233        assert!(check(
234            &[Requirement::new(Capability::FsRead, Some("/etc/passwd"))],
235            &p
236        )
237        .denied());
238        // unscoped requirement needs unscoped grant
239        assert!(check(&[Requirement::new(Capability::FsRead, None)], &p).denied());
240        assert!(check(&[Requirement::new(Capability::EnvRead, Some("HOME"))], &p).allowed);
241        assert!(check(&[Requirement::new(Capability::EnvRead, None)], &p).allowed);
242    }

244    #[test]
245    fn deny_wins_and_reasons_are_sorted() {
246        let mut p = policy();
247        p.grants.push(Grant::new(Capability::ShellExec, None));
248        let d = check(
249            &[
250                Requirement::with_evidence(Capability::ShellExec, None, "setup.sh:2"),
251                Requirement::with_evidence(Capability::FsWrite, Some("/x"), "setup.sh:3"),
252            ],
253            &p,
254        );
255        assert!(d.denied());
256        assert_eq!(d.reasons.len(), 2);
257        assert!(d
258            .reasons
259            .first()
260            .is_some_and(|r| r.starts_with("fs.write[/x] is not granted")));
261        assert!(d
262            .reasons
263            .get(1)
264            .is_some_and(|r| r.starts_with("shell.exec is denied")));
265    }

268    fn empty_requirements_are_allowed_by_any_policy() {

272    #[test]
273    fn policy_json_accepts_labels_and_objects() {
274        let p: Policy = serde_json::from_str(
275            r#"{"name":"p","grants":["fs.read[/a/*]",{"capability":"net.egress","scope":"x"},{"capability":"env.read"}],"denies":["shell.exec"]}"#,

277        .unwrap_or_default();

280            p.grants.first().map(Grant::label),
281            Some("fs.read[/a/*]".into())
282        );
283        assert_eq!(p.grants.get(2).map(Grant::label), Some("env.read".into()));
284        assert!(p.denies(Capability::ShellExec));
285        // a typo in a grant label grants nothing
286        let p: Policy =
287            serde_json::from_str(r#"{"name":"p","grants":["fs.raed"]}"#).unwrap_or_default();
288        assert!(check(&[Requirement::new(Capability::FsRead, None)], &p).denied());
289    }

291    fn arb_cap() -> impl Strategy<Value = Capability> {
292        (0..Capability::ALL.len()).prop_map(|i| {
293            Capability::ALL
294                .get(i)
295                .copied()
296                .unwrap_or(Capability::UiControl)
297        })
298    }

303            reqs in proptest::collection::vec((arb_cap(), proptest::option::of("[ab/]{0,4}")), 0..4),
304            grants in proptest::collection::vec((arb_cap(), proptest::option::of("[ab/*?]{0,4}")), 0..4),
305            denies in proptest::collection::vec(arb_cap(), 0..3),

307            let reqs: Vec<Requirement> = reqs.iter().map(|(c, s)| Requirement::new(*c, s.as_deref())).collect();
308            let policy = Policy { name: "p".into(), grants: grants.iter().map(|(c, s)| Grant::new(*c, s.as_deref())).collect(), denies };
309            let d = check(&reqs, &policy);
310            let expected = reqs.iter().all(|r| !policy.denies(r.capability) && policy.grants(r));
311            prop_assert_eq!(d.allowed, expected);
312            prop_assert_eq!(d.reasons.is_empty(), d.allowed);
313        }

318            grants in proptest::collection::vec((arb_cap(), proptest::option::of("[ab/*?]{0,4}")), 0..4),
319            extra in (arb_cap(), proptest::option::of("[ab/*?]{0,4}")),

321            let reqs: Vec<Requirement> = reqs.iter().map(|(c, s)| Requirement::new(*c, s.as_deref())).collect();
322            let mut policy = Policy { name: "p".into(), grants: grants.iter().map(|(c, s)| Grant::new(*c, s.as_deref())).collect(), denies: vec![] };
323            let before = check(&reqs, &policy).allowed;
324            policy.grants.push(Grant::new(extra.0, extra.1.as_deref()));
